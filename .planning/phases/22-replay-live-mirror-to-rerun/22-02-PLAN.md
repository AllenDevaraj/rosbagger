```python
def build_rerun_sink(rec, *, t0_ns=None):
    """Twin of build_publish_sink: deserialize each item independently and log to `rec`."""
    from rclpy.serialization import deserialize_message   # LAZY
    from rosidl_runtime_py.utilities import get_message   # LAZY
    from .converters import convert
    logged = {"n": 0}
    state = {"t0": t0_ns}
    classes = {}
    def rerun_sink(item) -> None:
        if state["t0"] is None: state["t0"] = item.t_ns
        if item.msgtype not in classes: classes[item.msgtype] = get_message(item.msgtype)
        msg = deserialize_message(item.cdr, classes[item.msgtype])
        rec.set_time("bag_time", duration=(item.t_ns - state["t0"]) / 1e9)
        try:
            for path, arch in convert(msg, item.msgtype, item.topic):
                rec.log(path, arch)
            logged["n"] += 1
        except Exception:   # one bad item must not abort the drive
            logged.setdefault("errors", 0); logged["errors"] += 1
    return rerun_sink, logged
```
