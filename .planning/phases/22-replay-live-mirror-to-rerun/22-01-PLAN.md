```python
def rerun_available() -> bool:
    """True iff rerun-sdk is importable (lazy probe; never raises)."""
    try:
        import rerun  # noqa: F401
    except ImportError:
        return False
    return True

def open_viewer(app_id: str = "rosbagger", *, save_path: str | None = None):
    """Build a RecordingStream: spawn the viewer (save_path=None) or write a .rrd.
    Lazy `import rerun` inside the body (offline invariant). 22-02 builds the sink that logs to it."""
    import rerun as rr
    rec = rr.RecordingStream(app_id)
    if save_path is not None:
        rec.save(save_path)
    else:
        rec.spawn()
    return rec
```

__init__.py (NEW): `from .session import rerun_available, open_viewer` (NO other module-top imports).
